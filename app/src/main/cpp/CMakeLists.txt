cmake_minimum_required(VERSION 3.22.1)
project(auralis_dsp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(auralis_dsp STATIC audio_engine.cpp bass_boost.cpp replaygain_tags.cpp transition_mixer.cpp safety_limiter.cpp)
target_include_directories(auralis_dsp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(auralis_dsp PRIVATE -Wall -Wextra -Wpedantic -Werror)

if(ANDROID)
    add_library(auralis SHARED jni_bridge.cpp transition_jni.cpp safety_limiter_jni.cpp)
    target_link_libraries(auralis PRIVATE auralis_dsp log android)
else()
    enable_testing()
    add_executable(dsp_reference_tests tests/dsp_reference_tests.cpp)
    target_link_libraries(dsp_reference_tests PRIVATE auralis_dsp)
    add_test(NAME dsp_reference_tests COMMAND dsp_reference_tests)
    add_executable(replaygain_tags_tests tests/replaygain_tags_tests.cpp)
    target_link_libraries(replaygain_tags_tests PRIVATE auralis_dsp)
    add_test(NAME replaygain_tags_tests COMMAND replaygain_tags_tests)
    add_executable(bass_boost_tests tests/bass_boost_tests.cpp)
    target_link_libraries(bass_boost_tests PRIVATE auralis_dsp)
    add_test(NAME bass_boost_tests COMMAND bass_boost_tests)
    add_executable(transition_mixer_tests tests/transition_mixer_tests.cpp)
    target_link_libraries(transition_mixer_tests PRIVATE auralis_dsp)
    add_test(NAME transition_mixer_tests COMMAND transition_mixer_tests)
    add_executable(safety_limiter_tests tests/safety_limiter_tests.cpp)
    target_link_libraries(safety_limiter_tests PRIVATE auralis_dsp)
    add_test(NAME safety_limiter_tests COMMAND safety_limiter_tests)
endif()
